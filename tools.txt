#!/bin/bash

start_date=$(date "+%Y%m%d_%H%M%S")
model_number=$(model modno)
serial_number=$(model serno)

file_name="/home/spectracom/log/dacStats_${model_number}_${serial_number}_${start_date}.csv"

first_line='datetime,zero_dac,dac,phase_error,freq_error'
echo "${first_line}" > "${file_name}"

#used for XO_ commands
SPEC_KTSAL_EN=true
export SPEC_KTSAL_EN

while :
do
    date_time=$(date "+%F %H:%M:%S")
    read -r dac _ < <(XO_GetDac 0)
    read -r zero_dac _ < <(XO_GetZeroDac 0)
    read -r phase _ < <(XO_GetPhaseError 0)
    read -r freq _ < <(XO_GetFreqError 0)
    echo "${date_time},0x${zero_dac},0x${dac},${phase},${freq}" >> "${file_name}"
    sleep 1
done
